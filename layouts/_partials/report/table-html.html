{{- /* Shared table markup for CSV data. Context: dict "page" "head" (columns to show, as written) "rows" "class". Severity / Risk / Confidence / Rating columns get colored pills; IP, host, port, hash, CVE and similar columns use the monospace font. Markdown in cells works. */ -}} {{- $page := .page }} {{- $pill := slice "severity" "risk" "rating" "confidence" "status" }} {{- $mono := slice "ip" "host" "port" "hash" "sha256" "sha-256" "sha1" "md5" "cve" "plugin id" "mac" "domain" "url" "wallet" "address" "indicator" "ioc" }}
{{- range .head }}{{ end }} {{- range $row := .rows }} {{- range $h := $.head }} {{- $k := lower $h }} {{- $v := index $row $k }} {{- if in $pill $k }} {{- $s := partial "report/sev.html" $v }} {{- else if in $mono $k }} {{- else }} {{- end }} {{- end }} {{- end }}
{{ . }}
