{{- /* Images dropped into a post (Obsidian drag-and-drop or paste): ![Caption shown under the image](evidence-1.png) An image on its own line becomes an evidence figure: resized to WebP, numbered "Fig. N" when it has a caption, and linked to the full-size file. Inline images stay inline. File names with spaces (Obsidian's "Pasted image 2026….png") work as-is. */ -}} {{- $u := urls.Parse .Destination }} {{- $path := strings.TrimPrefix "./" $u.Path }} {{- $img := false }} {{- if not $u.IsAbs }}{{ $img = .Page.Resources.Get $path }}{{ end }} {{- $alt := .PlainText }} {{- $src := .Destination }} {{- $full := .Destination }} {{- $w := 0 }}{{ $h := 0 }} {{- with $img }} {{- if and (eq .MediaType.MainType "image") (not (in (slice "svg" "gif") .MediaType.SubType)) }} {{- /* Re-encoded copies only: strips EXIF/GPS metadata. The original file is never published. */}} {{- $r := . }}{{ $big := . }} {{- if gt .Width 1360 }}{{ $r = .Resize "1360x webp q82" }}{{ else }}{{ $r = .Process "webp q82" }}{{ end }} {{- if gt .Width 2400 }}{{ $big = .Resize "2400x webp q85" }}{{ else }}{{ $big = .Process "webp q85" }}{{ end }} {{- $src = $r.RelPermalink }}{{ $full = $big.RelPermalink }}{{ $w = $r.Width }}{{ $h = $r.Height }} {{- else }} {{- $full = .RelPermalink }}{{ $src = .RelPermalink }} {{- end }} {{- end }} {{- if .IsBlock }} {{- $caption := .Title | default .Text }} {{- else -}} {{- end -}}
